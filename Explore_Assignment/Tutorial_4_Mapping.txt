#!/bin/bash
#SBATCH -J alignJH
#SBATCH -N 1
#SBATCH -n 1
#SBATCH -o output_%j.txt
#SBATCH -e error_%j.txt
#SBATCH --cpus-per-task=16
#SBATCH --mem=80G
#SBATCH --time=12:00:00

# Activate the STAR conda environment
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate /courses/BIOL2406.202710/shared/star_env

# Directory containing the genome index
GENOME_DIR="star_index"

# Directory containing the FASTQ files
FASTQ_DIR="fastq_files"

# Create output directory if it doesn't exist
mkdir -p aligned_output

# Loop through all paired-end FASTQ samples
for READ1 in "${FASTQ_DIR}"/SRR*_1.fastq; do
    SAMPLE=$(basename "$READ1" _1.fastq)

    # Define input files
    READ2="${FASTQ_DIR}/${SAMPLE}_2.fastq"

    # Align paired-end reads to the human reference genome
    STAR --runThreadN 16 \
         --genomeDir "${GENOME_DIR}" \
         --readFilesIn "$READ1" "$READ2" \
         --outFileNamePrefix "aligned_output/${SAMPLE}_" \
         --outSAMtype BAM SortedByCoordinate

done
